// Strip rich text flags before flattening (pdf-lib can't handle them) addLog('--- Starting rich text strip pass ---', 'info'); try { const allFields = form.getFields(); addLog(`Scanning ${allFields.length} fields for rich text...`, 'info'); allFields.forEach(field => { const name = field.getName(); const type = field.constructor.name; addLog(`Checking field: "${name}" (${type})`, 'info'); try { if (type === 'PDFTextField') { const rawField = field.acroField; addLog(` → acroField exists: ${!!rawField}`, 'info'); addLog(` → getFlags exists: ${!!rawField.getFlags}`, 'info'); const flags = rawField.getFlags ? rawField.getFlags() : 0; addLog(` → flags: ${flags}`, 'info'); const isRichText = (flags >> 25) & 1; addLog(` → isRichText: ${isRichText}`, 'info'); if (isRichText) { addLog(` → STRIPPING rich text from: "${name}"`, 'info'); if (rawField.setFlags) { rawField.setFlags(flags & ~(1 << 25)); addLog(` → setFlags done`, 'info'); } else { addLog(` → setFlags NOT available!`, 'error'); } addLog(` → deleting RV entry...`, 'info'); rawField.dict.delete(PDFLib.PDFName.of('RV')); addLog(` → RV deleted`, 'info'); } } } catch (fe) { addLog(` → ERROR on field "${name}": ${fe.message}`, 'error'); } }); } catch (prepErr) { addLog(`Rich text prep OUTER ERROR: ${prepErr.message}`, 'error'); } addLog('--- Rich text strip pass complete, calling flatten ---', 'info'); form.flatten(); addLog('--- flatten() completed ---', 'info');
